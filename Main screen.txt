<script>

function logout() {

    localStorage.removeItem(
        "currentUserId"
    );

    window.location.href =
        "login.html";
}

function loadDashboard() {

    const currentUserId =
        localStorage.getItem(
            "currentUserId"
        );

    if (!currentUserId) {

        window.location.href =
            "login.html";

        return;
    }

    const savedAccounts =
        JSON.parse(
            localStorage.getItem(
                "barangayAccounts"
            )
        ) || [];

    const account =
        savedAccounts.find(
            function(user) {

                return (
                    user.id === currentUserId &&
                    user.status === "active"
                );

            }
        );

    if (!account) {

        localStorage.removeItem(
            "currentUserId"
        );

        window.location.href =
            "login.html";

        return;
    }

    document
        .getElementById("dashboardName")
        .innerText =
        "Welcome, " +
        account.firstName +
        "!";

    document
        .getElementById("profileName")
        .innerText =
        account.firstName +
        " " +
        account.lastName;

    document
        .getElementById("profileBirthday")
        .innerText =
        account.birthday;

    document
        .getElementById("profileGender")
        .innerText =
        account.gender;

    document
        .getElementById("profileAddress")
        .innerText =
        account.address;

    document
        .getElementById("profilePhone")
        .innerText =
        account.phone;

    document
        .getElementById("profileEmail")
        .innerText =
        account.email;
}

loadDashboard();

</script>
